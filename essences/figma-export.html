export async function extractPalette( imageUrl: string, count = 5 ): Promise<string[]> { const img = new Image(); img.crossOrigin = "anonymous"; img.src = imageUrl; await new Promise(r => img.onload = r); const canvas = document.createElement("canvas"); const ctx = canvas.getContext("2d"); canvas.width = 50; canvas.height = 50; ctx.drawImage(img, 0, 0, 50, 50); // quantize pixel data const data = ctx.getImageData(0, 0, 50, 50); return quantize(data, count); }
