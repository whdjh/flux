```python
async def detect_rapid_change(data: HealthData):
    """10분 내 급격한 변화"""
    
    recent = get_last_10min_data(data.user_id, data.type)
    
    if data.type == "heart_rate":
        # 안정 상태인데 심박 급증
        if is_resting(user) and data.value > recent.avg + 20:
            # 이상 신호
            return await trigger_alert("heart_rate_spike")
    
    if data.type == "HRV":
        # 10분 만에 30% 하락
        if data.value < recent.avg * 0.7:
            # 저혈당 의심
            return await trigger_alert("possible_hypoglycemia")
```

**이런 경우만 즉시:**
```
✅ 심박 급증 (활동 없는데 +20)
✅ HRV 급락 (10분에 30%↓)
✅ SpO₂ 급락 (92% 미만)

→ 응급 상황 의심
→ 즉시 알림
```

```python
@shared_task
def morning_check():
    """아침 체크"""
    for user in get_active_users():
        # 어젯밤 데이터 분석
        sleep_data = get_last_night_sleep(user.id)
        today_calendar = get_today_schedule(user.id)
        
        # 오늘 위험 예측
        risk = await pattern_agent.analyze(
            sleep=sleep_data,
            schedule=today_calendar
        )
        
        if risk.level == "high":
            # 아침 알림
            send_notification(
                user.id,
                "⚠️ 오늘 주의 필요",
                risk.message
            )
```

```
        → 알림

22:00 - 저녁 체크
        → 오늘 피로도
        → 내일 일정 확인
```

```python
    """복약 기록 → 즉시 트리거"""

            )

@app.post("/api/calendar/event")
```

```python
@shared_task
def daily_pattern_analysis():
    """매일 새벽 2시 - 패턴 분석"""
    
    for user in get_active_users():
        # 7일 데이터 분석
        week_data = get_week_data(user.id)
        
        patterns = await pattern_agent.analyze_trends(
            hrv=week_data.hrv,
            sleep=week_data.sleep,
            activity=week_data.activity
        )
        
        # 악화 추세 감지
        if patterns.hrv_trend == "declining":
            if patterns.consecutive_days >= 3:
                # 3일 연속 하락
                flag_user_for_morning_alert(user.id)
        
        # 과부하 누적
        if patterns.overload_score > 70:
            # 번아웃 위험
            flag_burnout_risk(user.id)
        
        # 수면 부족 누적
        if patterns.sleep_debt > 10:  # 10시간 부족
            flag_sleep_recovery_needed(user.id)
```

**패턴 예시:**
```
HRV 하락 추세 (7일)
→ 3일 연속 ↓ → 주의
→ 5일 연속 ↓ → 경고
→ 7일 연속 ↓ → 강력 경고

수면 부족 누적
→ 부족 3일 → 회복 권장
→ 부족 5일 → 강력 휴식
→ 부족 7일 → 위험 상태

과부하 패턴
→ 고강도 운동 후 회복 부족
→ 3일 연속 반복
→ 번아웃 위험
```

```python
async def check_context_trigger(user_id: str):
    """맥락 종합 판단 - 가장 중요"""
    
    # 1. 현재 생체 데이터
    current = await get_current_vitals(user_id)
    
    # 2. 24시간 내 일정
    schedule = await get_upcoming_schedule(user_id, hours=24)
    
    # 3. 최근 복약
    medications = await get_recent_meds(user_id, days=1)
    
    # 4. 7일 패턴
    pattern = await get_week_pattern(user_id)
    
    # 5. 현재 시각/상황
    context = {
        "time": datetime.now(),
        "day_of_week": datetime.now().weekday(),
        "vitals": current,
        "schedule": schedule,
        "medications": medications,
        "pattern": pattern
    }
    
    # 6. 조합 판단
    risks = await orchestrator.evaluate_risks(context)
    
    return risks
```

**조합 예시:**

**Case 1: 새벽 운동 위험**
```
조건:
✓ HRV < P10
✓ 수면 < 6h
✓ 현재 시각 05:00-07:00
✓ 캘린더 "러닝" 06:00

→ 트리거 발동
→ "오늘 운동 위험" 알림
```

**Case 2: 수면제 + 다이빙**
```
조건:
✓ 복약: 졸피뎀 23:00

✓ 약물 잔존 가능

→ 즉시 트리거
```

**Case 3: 누적 과부하**

```
✓ 7일 평균 수면 < 6h

✓ 주말 등산 일정

→ 트리거 발동
→ "주말 휴식 권장"
```

**Case 4: 혈압약 미복용**

```
조건:
✓ 복약 시간 +2h 경과
```

## 📊 트리거 우선순위

```python
class TriggerPriority:
    CRITICAL = 1   # 생명 위험 (즉시)
    HIGH = 2       # 심각 위험 (당일)
    MEDIUM = 3     # 주의 필요 (1-2일)
    LOW = 4        # 정보성 (주간)

# 트리거 평가
async def evaluate_trigger(context):
    
    # Critical - 즉시 알림
    if has_drug_conflict(context):
        return Trigger(priority=CRITICAL, immediate=True)
    
    if detect_hypoglycemia(context):
        return Trigger(priority=CRITICAL, immediate=True)
    
    # High - 당일 알림
    if poor_recovery_before_exercise(context):
        return Trigger(priority=HIGH, timing="before_activity")
    
    # Medium - 예방적 알림
    if declining_trend(context):
        return Trigger(priority=MEDIUM, timing="morning")
    
    # Low - 주간 리포트
    if general_improvement_needed(context):
        return Trigger(priority=LOW, timing="weekly_report")
```

## 🎯 정리

### 질문: "데이터 변할 때만 트리거?"

**답: 아니다 ❌**

**트리거 발동 조건:**
```
1. 데이터 변화 (10%)
   - 급격한 변화만
   - 대부분은 저장만

2. 시간 (30%)
   - 정해진 시각
   - 일정 전

3. 이벤트 (20%)
   - 복약, 일정 입력
   - 사용자 행동

4. 패턴 (20%)
   - 누적 분석
   - 추세 감지

5. 조합 (20%)
   - 맥락 종합
   - 진짜 위험

→ 조합이 핵심!
```

**핵심:**
```
워치 데이터만으로는 부족
데이터 + 일정 + 복약 + 시간 + 패턴
= 전부 합쳐서 판단

단편적 트리거 = False Positive 많음
맥락적 트리거 = 정확도 ↑
```
