```bash
PROJECT_DIR="${PROJECT_PATH//\//-}"  # → -home-jun-8-August-electron

# 3. 최근 세션 파일 찾기
LATEST_SESSION=$(ls -t "$SESSIONS_DIR"/*.jsonl 2>/dev/null | head -1)

# 4. 세션 ID 추출
SESSION_ID=$(basename "$LATEST_SESSION" .jsonl)
# → 73f472cd-1e45-4150-9711-82f3468a7276

# 5. Short ID (처음 8자)
SHORT_ID="${SESSION_ID:0:8}"
# → 73f472cd
```

## 📝 2. 세션 제목 추출

### 방법 A: .jsonl 첫 메시지에서 추출

```python
import json

def get_session_title(session_file):
    """세션 제목 추출"""

    try:
        with open(session_file, 'r', encoding='utf-8') as f:
            # 첫 번째 라인 읽기
            first_line = f.readline()
            data = json.loads(first_line)

            # 사용자의 첫 메시지 찾기
            if 'content' in data:
                content = data['content']

                # 첫 줄만 가져오기 (제목으로 사용)
                first_sentence = content.split('\n')[0]

                # 너무 길면 자르기
                if len(first_sentence) > 50:
                    title = first_sentence[:50] + "..."
                else:
                    title = first_sentence

                return title

    except Exception as e:
        return None

    return "Untitled Session"

# 사용 예시
title = get_session_title("/home/jun/.claude/projects/.../73f472cd.jsonl")
# → "이전 세션이 날아가서, 이거 보고 내가 어떤거 하고있었는지..."
```

### 방법 B: Claude Code의 conversationTitle 필드 사용

```python
def get_conversation_title(session_file):
    """Claude Code가 생성한 대화 제목"""

    try:
        with open(session_file, 'r', encoding='utf-8') as f:
            for line in f:
                data = json.loads(line)

                # conversationTitle 필드 찾기
                if 'conversationTitle' in data:
                    return data['conversationTitle']

                # 또는 title 필드
                if 'title' in data:
                    return data['title']

    except Exception as e:
        pass

    # 없으면 방법 A로 폴백
    return get_session_title(session_file)
```

**권장**: 방법 B → 실패 시 방법 A → 실패 시 "Untitled Session"

## ⏰ 3. 시작 시간 추출

### 방법: 첫 TodoWrite 시점 또는 .jsonl 생성 시간

```python
def get_session_start_time(session_file):
    """세션 시작 시간"""

    # 방법 1: 파일 생성 시간
    import os
    from datetime import datetime

    creation_time = os.path.getctime(session_file)
    start_time = datetime.fromtimestamp(creation_time)

    return start_time.strftime("%Y-%m-%d %H:%M:%S")

# 또는

def get_first_message_time(session_file):
    """첫 메시지 시간"""

    try:
        with open(session_file, 'r', encoding='utf-8') as f:
            first_line = f.readline()
            data = json.loads(first_line)

            if 'timestamp' in data:
                return data['timestamp']

    except Exception as e:
        pass

    # 폴백: 파일 생성 시간
    return get_session_start_time(session_file)
```

## 🕐 4. 마지막 활동 시간 추출

### 방법: 현재 시간 또는 마지막 .jsonl 메시지

```python
def get_last_activity_time(session_file):
    """마지막 활동 시간"""

    # 방법 1: 파일 수정 시간 (가장 간단)
    import os
    from datetime import datetime

    mod_time = os.path.getmtime(session_file)
    last_time = datetime.fromtimestamp(mod_time)

    return last_time.strftime("%Y-%m-%d %H:%M:%S")

# 또는

def get_last_message_time(session_file):
    """마지막 메시지 시간"""

    try:
        with open(session_file, 'r', encoding='utf-8') as f:
            # 마지막 라인 읽기 (효율적인 방법)
            lines = f.readlines()
            if lines:
                last_line = lines[-1]
                data = json.loads(last_line)

                if 'timestamp' in data:
                    return data['timestamp']

    except Exception as e:
        pass

    # 폴백: 파일 수정 시간
    return get_last_activity_time(session_file)
```

## 📌 5. 세션 상태 판단

### 로직: 현재 시간과 마지막 활동 비교

```python
def get_session_status(last_activity_time):
    """세션 상태 판단"""

    from datetime import datetime, timedelta

    now = datetime.now()
    last_activity = datetime.strptime(last_activity_time, "%Y-%m-%d %H:%M:%S")

    # 30분 이상 활동 없으면 종료로 간주
    if now - last_activity > timedelta(minutes=30):
        return "closed"
    else:
        return "in_progress"
```

**기준**:
- 30분 이내 활동: `in_progress`
- 30분 이상 경과: `closed`

## 🔧 통합 함수

### 모든 메타데이터 한 번에 추출

```python
def extract_session_metadata():
    """세션 메타데이터 추출"""

    import os
    from datetime import datetime

    # 1. 프로젝트 경로
    project_path = os.getcwd()
    project_dir = project_path.replace('/', '-')

    # 2. 세션 파일 찾기
    sessions_dir = f"/home/jun/.claude/projects/{project_dir}"
    session_files = sorted(
        [f for f in os.listdir(sessions_dir) if f.endswith('.jsonl')],
        key=lambda f: os.path.getmtime(os.path.join(sessions_dir, f)),
        reverse=True
    )

    if not session_files:
        return None

    session_file = os.path.join(sessions_dir, session_files[0])

    # 3. 세션 ID
    session_id = session_files[0].replace('.jsonl', '')
    short_id = session_id[:8]

    # 4. 제목
    title = get_conversation_title(session_file)

    # 5. 시작 시간
    started = get_session_start_time(session_file)

    # 6. 마지막 활동
    last_activity = get_last_activity_time(session_file)

    # 7. 상태
    status = get_session_status(last_activity)

    return {
        'session_id': session_id,
        'short_id': short_id,
        'title': title,
        'started': started,
        'last_activity': last_activity,
        'status': status,
        'project_path': project_path
    }

# 사용 예시
metadata = extract_session_metadata()
print(metadata)
# {
#     'session_id': '73f472cd-1e45-4150-9711-82f3468a7276',
#     'short_id': '73f472cd',
#     'title': '이전 세션이 날아가서...',
#     'started': '2025-10-19 04:13:05',
#     'last_activity': '2025-10-19 04:17:11',
#     'status': 'closed',
#     'project_path': '/home/jun/8_August/electron'
# }
```

## 🔄 업데이트 로직

### 세션 파일이 이미 존재하는 경우

```python
def update_session_metadata(session_file, metadata):
    """세션 메타데이터 업데이트"""

    with open(session_file, 'r', encoding='utf-8') as f:
        content = f.read()

    # 기존 헤더 찾기
    import re

    # Last Activity 업데이트
    content = re.sub(
        r'\*\*Last Activity\*\*: .*',
        f"**Last Activity**: {metadata['last_activity']}",
        content
    )

    # Status 업데이트
    content = re.sub(
        r'\*\*Status\*\*: .*',
        f"**Status**: {metadata['status']}",
        content
    )

    with open(session_file, 'w', encoding='utf-8') as f:
        f.write(content)
```

## ⚠️ Edge Cases 처리

### 1. .jsonl 파일을 찾을 수 없음
```python
if not session_files:
    # Fallback: 타임스탬프 기반 ID 생성
    session_id = f"fallback-{datetime.now().strftime('%Y%m%d-%H%M%S')}"
    title = "Unknown Session"
```

### 2. 제목 추출 실패
```python
title = get_conversation_title(session_file) or "Untitled Session"
```

### 3. 시간 정보 없음
```python
started = get_session_start_time(session_file) or datetime.now().strftime("%Y-%m-%d %H:%M:%S")
```

### 4. 여러 프로젝트가 동시에 열림
