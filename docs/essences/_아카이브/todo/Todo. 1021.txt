### src/agents/shared/model-manager.ts

```typescript
import { ChatGoogleGenerativeAI } from '@langchain/google-genai';
import { MODEL_CONFIG, ModelName } from '../../config/models';

export class ModelManager {
  private static instances = new Map<ModelName, any>();
  
  static getModel(modelName: ModelName) {
    // 캐시된 인스턴스 반환
    if (this.instances.has(modelName)) {
      return this.instances.get(modelName);
    }
    
    const config = MODEL_CONFIG[modelName];
    
    if (!config) {
      throw new Error(`Unknown model: ${modelName}`);
    }
    
    if (config.provider === 'google') {
      const model = new ChatGoogleGenerativeAI({
        modelName: config.modelName,
        apiKey: process.env.GOOGLE_API_KEY!,
        temperature: config.temperature,
        maxOutputTokens: config.maxOutputTokens
      });
      
      this.instances.set(modelName, model);
      return model;
    }
    
    throw new Error(`Unsupported provider: ${config.provider}`);
  }
  
  static getAgentModel(agentType: string) {
    const envKey = `${agentType.toUpperCase()}_MODEL`;
    const modelName = (process.env[envKey] || 'gemini-flash') as ModelName;
    return this.getModel(modelName);
  }
}

export const PATTERN_ANALYSIS_PROMPT = `
당신은 건강 데이터 분석 전문가입니다.
사용자의 워치 데이터를 분석하여 위험도를 판단하세요.

# 입력 데이터
HRV: {hrv} (기준선: P10={baseline_hrv_p10}, P50={baseline_hrv_p50}, P90={baseline_hrv_p90})
심박수: {heart_rate} (기준선: P10={baseline_hr_p10}, P50={baseline_hr_p50}, P90={baseline_hr_p90})
수면: {sleep}시간 (기준선: P10={baseline_sleep_p10}, P50={baseline_sleep_p50})

# 분석 기준

RED (즉각 위험):
- HRV < P10 AND (수면 < 5.5h OR 심박수 > P90)
- HRV < P10 * 0.8 (극심한 저하)
- 생명 위험 가능성

AMBER (높은 위험):
- HRV < P20 AND 수면 < 6h
- HRV < P50 * 0.7
- 당일 조치 필요

GREEN (정상):
- 나머지 경우
- 정상 범위

# 출력 형식
반드시 아래 JSON 형식으로만 응답하세요. 다른 텍스트는 포함하지 마세요.

{{
  "alertLevel": "RED" | "AMBER" | "GREEN",
  "patterns": ["구체적 패턴 설명1", "구체적 패턴 설명2"],
  "confidence": 0.0-1.0,
  "reasoning": "판단 근거를 명확히 설명",
  "data": {{
    "hrv": {{
      "current": {hrv},
      "baseline": {baseline_hrv_p50},
      "deviation": -38.2
    }},
    "sleep": {{
      "current": {sleep},
      "baseline": {baseline_sleep_p50},
      "deviation": -31.1
    }},
    "heartRate": {{
      "current": {heart_rate},
      "baseline": {baseline_hr_p50},
      "deviation": 37.0
    }}
  }}
}}
`;

### src/agents/pattern/index.ts

```typescript
import { ModelManager } from '../shared/model-manager';
import { HealthData, UserBaseline } from '../../types/health';
import { PatternAnalysis } from '../../types/agent';
import { PATTERN_ANALYSIS_PROMPT } from './prompts';

export class PatternAgent {
  private llm: any;
  
  constructor() {
    this.llm = ModelManager.getAgentModel('pattern');
  }
  
  async analyze(
    healthData: HealthData,
    baseline: UserBaseline
  ): Promise<PatternAnalysis> {
    try {
      // 프롬프트 생성
      const prompt = PATTERN_ANALYSIS_PROMPT
        .replace('{hrv}', healthData.hrv.toString())
        .replace('{heart_rate}', healthData.heartRate.toString())
        .replace('{sleep}', healthData.sleep.toString())
        .replace('{baseline_hrv_p10}', baseline.hrv.p10.toString())
        .replace('{baseline_hrv_p50}', baseline.hrv.p50.toString())
        .replace('{baseline_hrv_p90}', baseline.hrv.p90.toString())
        .replace('{baseline_hr_p10}', baseline.heartRate.p10.toString())
        .replace('{baseline_hr_p50}', baseline.heartRate.p50.toString())
        .replace('{baseline_hr_p90}', baseline.heartRate.p90.toString())
        .replace('{baseline_sleep_p10}', baseline.sleep.p10.toString())
        .replace('{baseline_sleep_p50}', baseline.sleep.p50.toString());
      
      // LLM 호출
      const response = await this.llm.invoke(prompt);
      
      // JSON 파싱
      let content = response.content as string;
      
      // Gemini가 가끔 ```json ... ``` 로 감싸는 경우 처리
      content = content
        .replace(/```json\n?/g, '')
        .replace(/```\n?/g, '')
        .trim();
      
      const result = JSON.parse(content);
      
      return result;
      
    } catch (error) {
      console.error('Pattern Agent Error:', error);
      throw error;
    }
  }
}

### src/agents/coaching/index.ts

```typescript
import { ModelManager } from '../shared/model-manager';
import { PatternAnalysis, CoachingResult } from '../../types/agent';
import { CalendarEvent } from '../../types/health';
import { COACHING_PROMPT } from './prompts';

export class CoachingAgent {
  private llm: any;
  
  constructor() {
    this.llm = ModelManager.getAgentModel('coaching');
  }
  
  async generate(
    patternAnalysis: PatternAnalysis,
    calendarEvent?: CalendarEvent
  ): Promise<CoachingResult> {
    try {
      // 캘린더 이벤트 문자열화
      const eventStr = calendarEvent 
        ? `${calendarEvent.title} (${calendarEvent.startTime.toLocaleTimeString('ko-KR', { hour: '2-digit', minute: '2-digit' })})`
        : '없음';
      
      // 프롬프트 생성
      const prompt = COACHING_PROMPT
        .replace('{alert_level}', patternAnalysis.alertLevel)
        .replace('{patterns}', patternAnalysis.patterns.join(', '))
        .replace('{reasoning}', patternAnalysis.reasoning)
        .replace('{hrv_current}', patternAnalysis.data.hrv.current.toString())
        .replace('{hrv_baseline}', patternAnalysis.data.hrv.baseline.toString())
        .replace('{hrv_deviation}', patternAnalysis.data.hrv.deviation.toFixed(1))
        .replace('{sleep_current}', patternAnalysis.data.sleep.current.toString())
        .replace('{sleep_baseline}', patternAnalysis.data.sleep.baseline.toString())
        .replace('{sleep_deviation}', patternAnalysis.data.sleep.deviation.toFixed(1))
        .replace('{hr_current}', patternAnalysis.data.heartRate.current.toString())
        .replace('{hr_baseline}', patternAnalysis.data.heartRate.baseline.toString())
        .replace('{hr_deviation}', patternAnalysis.data.heartRate.deviation.toFixed(1))
        .replace('{calendar_event}', eventStr);
      
      // LLM 호출
      const response = await this.llm.invoke(prompt);
      
      // JSON 파싱
      let content = response.content as string;
      content = content
        .replace(/```json\n?/g, '')
        .replace(/```\n?/g, '')
        .trim();
      
      const result = JSON.parse(content);
      
      return result;
      
    } catch (error) {
      console.error('Coaching Agent Error:', error);
      throw error;
    }
  }
}

// 간단한 약물 DB (실제로는 외부 DB 사용)
export interface DrugInfo {
  name: string;
  genericName: string;
  duration: number; // 시간
  contraindications: string[];
  activities: string[];
}

export const DRUG_DATABASE: Record<string, DrugInfo> = {
  '졸피뎀': {
    name: '졸피뎀',
    genericName: 'Zolpidem',
    duration: 10,
    contraindications: ['중추신경억제'],
    activities: ['수중활동', '다이빙', '운전', '기계조작']
  },
  '스틸녹스': {
    name: '스틸녹스',
    genericName: 'Zolpidem',
    duration: 10,
    contraindications: ['중추신경억제'],
    activities: ['수중활동', '다이빙', '운전', '기계조작']
  },
  '아스피린': {
    name: '아스피린',
    genericName: 'Aspirin',
    duration: 6,
    contraindications: ['혈액응고장애'],
    activities: ['격렬한운동', '접촉스포츠']
  }
};

export function findDrug(name: string): DrugInfo | undefined {
  // 정확한 이름 매칭
  if (DRUG_DATABASE[name]) {
    return DRUG_DATABASE[name];
  }
  
  // 부분 매칭
  for (const [key, drug] of Object.entries(DRUG_DATABASE)) {
    if (name.includes(key) || key.includes(name)) {
      return drug;
    }
  }
  
  return undefined;
}

```typescript
import { ModelManager } from '../shared/model-manager';
import { MedicalCheck } from '../../types/agent';
import { Medication, CalendarEvent } from '../../types/health';
import { findDrug } from './drug-database';
import { MEDICAL_CHECK_PROMPT } from './prompts';

export class MedicalContextAgent {
  private llm: any;
  
  constructor() {
    this.llm = ModelManager.getAgentModel('medical');
  }
  
  async check(
    medication: Medication,
    calendarEvent: CalendarEvent
  ): Promise<MedicalCheck> {
    try {
      // 약물 DB 조회
      const drugInfo = findDrug(medication.name);
      
      if (!drugInfo) {
        // 약물 정보 없으면 LLM에게 물어보기
        return this.checkWithLLM(medication, calendarEvent);
      }
      
      // 시간 차이 계산
      const timeDiff = (calendarEvent.startTime.getTime() - medication.takenAt.getTime()) / (1000 * 60 * 60);
      
      // 규칙 기반 체크
      if (timeDiff < drugInfo.duration) {
        // 금기 활동인지 확인
        const activityLower = calendarEvent.title.toLowerCase();
        const isContraindicated = drugInfo.activities.some(activity => 
          activityLower.includes(activity)
        );
        
        if (isContraindicated) {
          return {
            safe: false,
            warning: {
              level: 'RED',
              message: `${medication.name} + ${calendarEvent.title} 매우 위험!`,
              reason: `${medication.name}은 최대 ${drugInfo.duration}시간 동안 체내에 잔존하며, ${drugInfo.contraindications.join(', ')} 효과가 있습니다. ${calendarEvent.title} 활동은 위험합니다.`,
              alternatives: [
                `복약 취소`,
                `${calendarEvent.title} 연기 (${drugInfo.duration + 2}시간 후)`,
                `대체 활동 선택`
              ]
            }
          };
        }
      }
      
      if (timeDiff >= drugInfo.duration && timeDiff < drugInfo.duration + 2) {
        return {
          safe: false,
          warning: {
            level: 'AMBER',
            message: `${medication.name} 효과가 완전히 사라지지 않았습니다`,
            reason: `약물 지속 시간 종료 후 2시간 이내입니다. 주의가 필요합니다.`,
            alternatives: [
              `2시간 더 기다리기`,
              `저강도로 진행`
            ]
          }
        };
      }
      
      return { safe: true };
      
    } catch (error) {
      console.error('Medical Context Agent Error:', error);
      throw error;
    }
  }
  
  private async checkWithLLM(
    medication: Medication,
    calendarEvent: CalendarEvent
  ): Promise<MedicalCheck> {
    // LLM에게 물어보기 (약물 DB에 없을 때)
    const timeDiff = (calendarEvent.startTime.getTime() - medication.takenAt.getTime()) / (1000 * 60 * 60);
    
    const prompt = `
약물: ${medication.name}
복용 시간: ${medication.takenAt.toLocaleString('ko-KR')}
활동: ${calendarEvent.title}
활동 시간: ${calendarEvent.startTime.toLocaleString('ko-KR')}
시간 차이: ${timeDiff.toFixed(1)}시간

이 약물과 활동의 충돌 여부를 판단하세요.
JSON 형식으로 응답:
{
  "safe": true | false,
  "warning": {
    "level": "RED" | "AMBER",
    "message": "경고 메시지",
    "reason": "이유",
    "alternatives": ["대안"]
  }
}
`;
    
    const response = await this.llm.invoke(prompt);
    
    let content = response.content as string;
    content = content
      .replace(/```json\n?/g, '')
      .replace(/```\n?/g, '')
      .trim();
    
    return JSON.parse(content);
  }
}

## 10. LangGraph 워크플로우

### src/workflow/nodes.ts

```typescript
import { AgentState } from '../types/state';
import { PatternAgent } from '../agents/pattern';
import { CoachingAgent } from '../agents/coaching';

// Node 함수들
export async function patternAnalysisNode(
  state: AgentState
): Promise<Partial<AgentState>> {
  console.log('🔍 Pattern Agent 실행...');
  
  const agent = new PatternAgent();
  
  const analysis = await agent.analyze(
    state.healthData,
    state.baseline
  );
  
  console.log(`  ✅ 분석 완료: ${analysis.alertLevel}`);
  
  return {
    patternAnalysis: analysis
  };
}

export async function coachingNode(
  state: AgentState
): Promise<Partial<AgentState>> {
  console.log('💪 Coaching Agent 실행...');
  
  if (!state.patternAnalysis) {
    throw new Error('Pattern analysis required');
  }
  
  const agent = new CoachingAgent();
  
  const result = await agent.generate(
    state.patternAnalysis,
    state.calendarEvent
  );
  
  console.log(`  ✅ 조언 생성 완료`);
  
  return {
    coachingResult: result
  };
}

// 조건부 라우팅
export function shouldSendNotification(state: AgentState): string {
  const level = state.patternAnalysis?.alertLevel;
  
  if (level === 'RED' || level === 'AMBER') {
    return 'coaching';
  }
  
  return 'end';
}
```

### src/workflow/graph.ts

```typescript
import { StateGraph, END } from '@langchain/langgraph';
import { AgentState } from '../types/state';
import {
  patternAnalysisNode,
  coachingNode,
  shouldSendNotification
} from './nodes';

// State 채널 정의
const channels = {
  userId: null,
  trigger: null,
  timestamp: null,
  healthData: null,
  baseline: null,
  recentHistory: null,
  calendarEvent: null,
  medication: null,
  patternAnalysis: null,
  coachingResult: null,
  goalProposal: null,
  medicalCheck: null,
  error: null,
  processingTime: null
};

// 워크플로우 그래프 생성
export function createHealthAgentGraph() {
  const workflow = new StateGraph<AgentState>({ channels });
  
  // 노드 추가
  workflow.addNode('pattern_analysis', patternAnalysisNode);
  workflow.addNode('coaching', coachingNode);
  
  // 엣지 추가
  workflow.addEdge('__start__', 'pattern_analysis');
  
  // 조건부 엣지
  workflow.addConditionalEdges(
    'pattern_analysis',
    shouldSendNotification,
    {
      'coaching': 'coaching',
      'end': END
    }
  );
  
  workflow.addEdge('coaching', END);
  
  // 컴파일
  return workflow.compile();
}

export const healthAgentGraph = createHealthAgentGraph();
```

      console.log(`    - ${p}`);

    console.log(`  근거: ${result.patternAnalysis.reasoning}\n`);

  if (result.coachingResult) {
    console.log('💬 Coaching Result:');
    console.log(`  ${result.coachingResult.headline}`);
    console.log(`\n  데이터:`);

      console.log(`    • ${d}`);
    });
    console.log(`\n  이유: ${result.coachingResult.detail.reasoning}`);
    console.log(`\n  제안: ${result.coachingResult.detail.suggestion}`);
    console.log(`\n  액션:`);
    result.coachingResult.actions.forEach((a: any) => {
      console.log(`    [${a.label}]`);
    });
  }
  
  if (!result.coachingResult) {
    console.log('✅ GREEN: 정상 상태, 알림 없음');
  }
}

main().catch(console.error);
